# SubDSculpt: drag one SubD control vertex with a falloff (the "SCU_" section, tests/sections/79_sculpt.sh).
#
# Fixture: the app's SubD box 0,0,0 - 10,10,0 - 10 (the plain 8-vertex cube cage). The vertex at the origin is dragged to (0,0,5), i.e. by
# (0,0,5), with a Radius and a Falloff; every control point within Radius of the ORIGINAL origin moves by weight * (0,0,5) (weights in the
# section header). Each case then reads back the dragged vertex (limit point) and one of its edge neighbours (10,0,z) with SubDVertexInfo.
# smoke.sh runs this script; there is no @TMP@ substitution here.

SetActiveViewport Top

# Constant, radius 11: the dragged vertex and its three edge neighbours (distance 10) move by the full (0,0,5); the three face-diagonal vertices
# (distance 14.142) and the far corner stay
New
SCUBEGIN_constant
SubDBox 0,0,0 10,10,0 10
SelLast
SubDSculpt
Radius=11
Falloff=Constant
0,0,0
0,0,5
SelLast
SubDVertexInfo
0,0,5
10,0,5
Enter

# Linear, radius 15: weights 1 (picked), 1 - 10/15 = 1/3 (edge neighbours), 1 - 14.142/15 = 0.05719 (face diagonals), 0 (far corner, 17.32)
New
SCUBEGIN_linear
SubDBox 0,0,0 10,10,0 10
SelLast
SubDSculpt
Radius=15
Falloff=Linear
0,0,0
0,0,5
SelLast
SubDVertexInfo
0,0,5
10,0,1.7
Enter

# Smooth (smoothstep), radius 15
New
SCUBEGIN_smooth
SubDBox 0,0,0 10,10,0 10
SelLast
SubDSculpt
Radius=15
Falloff=Smooth
0,0,0
0,0,5
SelLast
SubDVertexInfo
0,0,5
10,0,1.3
Enter

# radius 5 (below the edge length 10): only the dragged vertex moves
New
SCUBEGIN_single
SubDBox 0,0,0 10,10,0 10
SelLast
SubDSculpt
Radius=5
Falloff=Smooth
0,0,0
0,0,5
SelLast
SubDVertexInfo
0,0,5
10,0,0
Enter

# negative control: a first pick far from every vertex does nothing
New
SCUBEGIN_miss
SubDBox 0,0,0 10,10,0 10
SelLast
SubDSculpt
Radius=15
0,0,-40
0,0,5
SelLast
SubDVertexInfo
0,0,0
Enter

# Constrain=Normal: dragging the origin corner towards (-3,0,0) counts only the part along its vertex normal (-1,-1,-1)/sqrt 3:
# (-3,0,0) . n = 3/sqrt 3 = sqrt 3, so the vertex moves by n * sqrt 3 = (-1,-1,-1) (a free drag would move it to (-3,0,0)). Radius 5: only that vertex.
New
SCUBEGIN_normal
SubDBox 0,0,0 10,10,0 10
SelLast
SubDSculpt
Radius=5
Constrain=Normal
0,0,0
-3,0,0
SelLast
SubDVertexInfo
-1,-1,-1
Enter
